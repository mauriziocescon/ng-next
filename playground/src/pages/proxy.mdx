## Proxy Forwarding

<p class="subtitle"><code>component.proxy&lt;T&gt;()</code> exposes a native element surface — directives forward at compile time through <code>@forward()</code>.</p>

Directives applied to the component are type-checked against the proxy's element type, then placed on the `@forward()` target inside the template.

```typescript
import { component, input, output, computed, fragment, signal } from '@angular/core';
import { ripple } from '@mylib/ripple';
import { tooltip } from '@mylib/tooltip';

// -- Definition: proxy surface --
export const Button = component.proxy<HTMLButtonElement>({
  bindings: {
    type: input<'button' | 'submit' | 'reset'>('button'),
    disabled: input<boolean>(false),
    click: output<void>(),
    children: fragment.required<void>(),
  },
  setup: ({ type, disabled, click, children }) => @{
    <button
      @forward()
      type={type()}
      disabled={disabled()}
      on:click={() => click.emit()}>
      @render(children())
    </button>
  },
});

// -- Usage: directives pass through --
export const Consumer = component({
  setup: () => {
    const tooltipMsg = signal('Hover me');
    const valid = signal(false);

    return @{
      <Button
        type="button"
        use:ripple()
        use:tooltip(message={tooltipMsg()})
        disabled={!valid()}
        on:click={() => {}}>
          Click me
      </Button>
    };
  },
});
```

<div class="annotation" role="note">
  <strong>No runtime spread:</strong> the compiler expands <code>@forward()</code> into concrete bindings and directive instructions. <code>use:ripple()</code> and <code>use:tooltip(...)</code> are accepted because their <code>host</code> types are compatible with <code>HTMLButtonElement</code>. The same directive cannot appear twice on the same final element.
</div>
