## Composition with Fragments

<p class="subtitle">What if ng-content and ng-template were unified into typed functions?</p>

Fragments are similar to [Svelte snippets](https://svelte.dev/docs/svelte/snippet): functions that return HTML markup. The returned markup is opaque — it cannot be manipulated like React Children or Solid children.

### Implicit children fragment

Markup inside a component tag implicitly becomes a `children` fragment. `children` is a reserved name — provided by Angular from nested content, not directly bindable.

```typescript
import { component, signal } from '@angular/core';
import { Menu, MenuItem } from '@mylib/menu';

export const MenuConsumer = component({
  setup: () => {
    const first = signal('First');
    const second = signal('Second');

    return @{
      <Menu>
        <MenuItem>{first()}</MenuItem>
        <MenuItem>{second()}</MenuItem>
      </Menu>
    };
  },
});
```

### Menu component using fragments

```typescript
import { component, fragment } from '@angular/core';

export const Menu = component({
  bindings: {
    // Provided by Angular from nested content (not bindable directly). Reserved name.
    children: fragment<void>(),
  },
  setup: ({ children }) => {
    return @{
      @if (children) {
        @render(children())
      } @else {
        <span>Empty</span>
      }
    };
  },
});
```

### Custom fragment slots

Inline `@fragment` declarations are auto-passed as the matching fragment input by name.

```typescript
import { component, signal, input, fragment } from '@angular/core';
import { Menu } from '@mylib/menu';
import { MyMenuItem } from './my-menu-item.ng';

export interface Item { id: string; desc: string; }

export const MenuConsumer = component({
  setup: () => {
    const items = signal<Item[]>(/** ... **/);

    return @{
      <Menu items={items()}>
        @fragment menuItem(item: Item) {
          <div class="my-menu-item">
            <MyMenuItem>{item.desc}</MyMenuItem>
          </div>
        }
      </Menu>
    };
  },
});
```

<div class="annotation">
  <strong>Rendering:</strong> <code>@render(fragment())</code> renders a fragment. No <code>ng-container</code> needed. The full form supports an optional injector: <code>@render(fragment(), {'{ injector }'})</code>.
</div>
