## Composition with Fragments

<p class="subtitle">What if ng-content and ng-template were unified into typed functions?</p>

Fragments are typed functions that return markup — implicit for children, named for custom slots.

```typescript
import { component, signal, input, fragment } from '@angular/core';
import { MyMenuItem } from './my-menu-item.ng';

export interface Item { id: string; desc: string; }

export const MenuConsumer = component({
  setup: () => {
    const items = signal<Item[]>(/** ... **/);

    return @{
      <Menu items={items()}>
        @fragment menuItem(item: Item) {
          <div class="my-menu-item">
            <MyMenuItem>{item.desc}</MyMenuItem>
          </div>
        }
      </Menu>
    };
  },
});

// -- Menu --
export const Menu = component({
  bindings: {
    items: input.required<{ id: string; desc: string }[]>(),
    menuItem: fragment.required<[{ id: string; desc: string }]>(),
    children: fragment<void>(),
  },
  setup: ({ items, menuItem, children }) => @{
    <h1>Total items: {items().length}</h1>

    @for (item of items(); track item.id) {
      @render(menuItem(item))
    }

    @if (children) {
      @render(children())
    }
  },
});
```

<div class="annotation">
  <strong>One concept, two modes:</strong> nested markup becomes an implicit <code>children</code> fragment; named <code>@fragment</code> declarations are auto-matched to typed fragment inputs. No <code>ng-content</code>, no <code>ng-template</code>, no <code>let-*</code>.
</div>
