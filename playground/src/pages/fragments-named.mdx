## Fragments: Named

<p class="subtitle">Named fragments are typed template functions — custom slots with parameters, auto-matched by name.</p>

A `@fragment` declaration inside a component tag is matched to a `fragment` binding by name. The consuming component calls it with typed arguments.

```typescript
import { component, signal, input, fragment } from '@angular/core';
import { MyMenuItem } from './my-menu-item.ng';

export interface Item { id: string; desc: string; }

// -- Consumer: declares a named fragment inline --
export const Consumer = component({
  setup: () => {
    const items = signal<Item[]>(/** ... **/);

    return @{
      <Menu items={items()}>
        @fragment menuItem(item: Item) {
          <div class="my-menu-item">
            <MyMenuItem>{item.desc}</MyMenuItem>
          </div>
        }
      </Menu>
    };
  },
});

// -- Menu: accepts a typed fragment input --
export const Menu = component({
  bindings: {
    items: input.required<{ id: string; desc: string }[]>(),
    menuItem: fragment.required<[{ id: string; desc: string }]>(),
  },
  setup: ({ items, menuItem }) => @{
    <h1>Total items: {items().length}</h1>

    @for (item of items(); track item.id) {
      @render(menuItem(item))
    }
  },
});
```

<div class="annotation" role="note">
  <strong>Type-safe slots:</strong> the fragment's parameter list is typed at the declaration site. The component calls <code>menuItem(item)</code> — the compiler checks the argument matches <code>[{'{'}id: string; desc: string{'}'}]</code>. No <code>let-*</code>, no <code>ng-template</code>, no <code>ngTemplateGuard</code>.
</div>
