## Component Root: Native Element

<p class="subtitle"><code>rootNode: element&lt;T&gt;()</code> declares a native-element root — directives applied to the component attach to that root at compile time.</p>

Declaring `rootNode: element<T>()` gives a component a single native-element root — one native root node, not wrapped in control flow, a `@defer` block, or a `@boundary` block. Call-site directives attach to that element, accepted only when their `host` type accepts it.

```typescript
import { component, input, output, computed, fragment, signal, element } from '@angular/core';
import { ripple } from '@mylib/ripple';
import { tooltip } from '@mylib/tooltip';

export const Consumer = component({
  setup: () => {
    const tooltipMsg = signal('Hover me');
    const valid = signal(false);

    return @{
      <Button
        type="button"
        use:ripple()
        use:tooltip(message={tooltipMsg()})
        disabled={!valid()}
        on:click={() => {}}>
          Click me
      </Button>
    };
  },
});

export const Button = component({
  rootNode: element<HTMLButtonElement>(),
  bindings: {
    type: input<'button' | 'submit' | 'reset'>('button'),
    disabled: input<boolean>(false),
    click: output<void>(),
    children: fragment.required<void>(),
  },
  setup: ({ type, disabled, click, children }) => @{
    <button
      type={type()}
      disabled={disabled()}
      on:click={() => click.emit()}>
      @render(children()())
    </button>
  },
});
```

<div class="annotation" role="note">
  <strong>Directives land on the root:</strong> because <code>Button</code> declares <code>rootNode: element&lt;HTMLButtonElement&gt;()</code>, its single <code>&lt;button&gt;</code> root is where call-site directives attach. <code>use:ripple()</code> and <code>use:tooltip(...)</code> are accepted because their <code>host</code> types accept <code>HTMLButtonElement</code>.
</div>

<div class="annotation" role="note">
  <strong>No duplicate directives:</strong> the same directive cannot appear twice on the same resolved element — a call-site <code>use:tooltip(...)</code> collides with one already written on the root inside <code>Button</code>.
</div>
